      // majority vote: 3 fixed samples, each guard-checked (on-topic, WHY
      // non-empty), then resolveDecisionVote. Asymmetry, disclosed in the
      // write-up: flipped samples are rejected before voting, so a
      // hold-pattern question can only resolve hold-side or abstain — never
      // continue-side. Safe direction: an honest abstention beats a wrong
      // call. Other questions keep first-accept (max 3 attempts).
      let accepted = null, lastRaw = '';
      const isHoldPattern = /whether to (put|place|keep).* (on hold|pause|postpone|defer)/i.test(inputs.decision);
      if (isHoldPattern) {
        const samples = [];
        for (let attempt = 0; attempt < 3; attempt++) {
          const rawD = await generate(SYS_DECISION,
            `Input: Decision I'm putting off: ${inputs.decision}. Context: ${inputs.mind || 'none given'}.\nOutput:`, 120);
          lastRaw = rawD;
          const parsed = parseDecision(rawD);
          if (parsed && parsed.why && isOnTopicDecision(parsed.choice, inputs.decision)) samples.push(parsed);
        }
        accepted = resolveDecisionVote(samples);
      } else {
        for (let attempt = 0; attempt < 3 && !accepted; attempt++) {
          const rawD = await generate(SYS_DECISION,
            `Input: Decision I'm putting off: ${inputs.decision}. Context: ${inputs.mind || 'none given'}.\nOutput:`, 120);
          lastRaw = rawD;
          const parsed = parseDecision(rawD);
          if (parsed && parsed.why && isOnTopicDecision(parsed.choice, inputs.decision)) accepted = parsed;
        }
      }
      current.raw.decision = lastRaw;
      current.decision = accepted;
      current.decisionUndecided = !accepted;
    }
    renderLog(6, -1);
    current.ts = Date.now();
    localStorage.setItem(LS_KEY, JSON.stringify(current));
    renderBrief();
    setStatus('Brief ready. Total bytes of your private life that left this device: 0.');
  } catch (err) {
    console.error(err);
    if (String(err && err.message) === 'garbage-tokens') {
      setStatus('The model produced garbage tokens \u2014 reload the page and let it take the fp32 path again.');
    } else {
      setStatus('Something went wrong on the night shift \u2014 check the console and try again.');
    }
  } finally {
    goBtn.disabled = false;
    $('demo').disabled = false;
  }
}
