    // Guard gave up after retries: say so plainly, never dress a wrong
    // decision as the call. The wrong raw takes are NOT shown as the decision.
    $('sec-decision').hidden = false;
    $('b-decision').innerHTML = '<span style="color:var(--muted2)">Your chief of staff couldn\u2019t settle this one \u2014 the model kept answering a different question, so it left this undecided rather than guessing.</span>';
  } else if (b.inputs.decision && b.raw.decision) {
    // Same silent-failure class as the replies fix: the model answered but
    // wandered off-format — show its raw take instead of dropping the decision.
    $('sec-decision').hidden = false;
    $('b-decision').innerHTML = '<span style="color:var(--muted2)">The model wandered off-format — here is its raw take:</span><br/>' +
      esc(b.raw.decision).replace(/\n/g, '<br/>');
  } else {
    $('sec-decision').hidden = true;
  }
  $('brief').scrollIntoView({ behavior: 'smooth', block: 'start' });
}

function briefToMarkdown() {
  const b = current;
  const lines = ['# \u2600\uFE0F Morning brief \u2014 ' + new Date(b.ts).toLocaleDateString(), '', '## Top 3 today'];
  if (b.priorities.length) b.priorities.forEach((p, i) => lines.push(`${i + 1}. ${p}`));
  else if (b.raw.priorities) lines.push(b.raw.priorities);
  else lines.push('- (nothing to prioritise tonight)');
  lines.push('', '## Drafted replies');
  if (b.replies.length) b.replies.forEach(r => lines.push(`- **${r.who}:** ${r.text}`));
  else lines.push('- (nothing needs an answer)');
  if (b.decision) {
    lines.push('', '## Decided for you', `**${b.decision.choice}**${b.decision.why ? ' \u2014 ' + b.decision.why : ''}`);
  } else if (b.decisionUndecided) {
    lines.push('', '## Decided for you', '(left undecided \u2014 the model kept answering a different question, so no call was made)');
  } else if (b.raw.decision) {
    lines.push('', '## Decided for you (model went off-format — raw take)', b.raw.decision);
  }
  return lines.join('\n');
}
