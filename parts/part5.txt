    const m = line.match(/^REPLY TO\s+(.+?)\s*:\s*(.+)$/i);
    if (m) out.push({ who: m[1].trim(), text: m[2].trim() });
  }
  return out.slice(0, 3);
}
function parseDecision(text) {
  let choice = '', why = '';
  for (const line of text.split('\n')) {
    let m = line.match(/^DECISION\s*:\s*(.+)$/i); if (m) choice = m[1].trim();
    m = line.match(/^WHY\s*:\s*(.+)$/i); if (m) why = m[1].trim();
  }
  return choice ? { choice, why } : null;
}

// ---------- Night log ----------
const LOG_STEPS = [
  'Read tomorrow\u2019s schedule',
  'Sorted the day \u2014 picked your top 3',
  'Triaged the unread messages',
  'Drafted replies in your voice',
  'Made the call you were putting off',
  'Brief brewed. Sleep well. \u2600\uFE0F'
];
function renderLog(doneCount, activeIdx) {
  const ul = $('log-list');
  ul.innerHTML = '';
  LOG_STEPS.forEach((label, i) => {
    const inp = current && current.inputs;
    if (i === 1 && !(inp && (inp.schedule || inp.mind))) return; // nothing to sort, no step
    if ((i === 2 || i === 3) && !(inp && inp.messages)) return; // no messages, no steps
    if (i === 4 && !(inp && inp.decision)) return; // no decision asked, no step
    const li = document.createElement('li');
    if (i < doneCount) li.className = 'done';
    const box = document.createElement('span');
    box.className = 'box';
    box.textContent = i < doneCount ? '\u2611\uFE0F' : (i === activeIdx ? '\u231B' : '\u2610');
    li.append(box, document.createTextNode(label));
    ul.append(li);
  });
  $('nightlog').hidden = false;
}
